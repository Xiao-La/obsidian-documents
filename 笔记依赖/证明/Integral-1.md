
$$
\int \sin x\,\mathrm{d}x=-\cos x+C
$$
$$
\int \cos x\,\mathrm{d}x=\sin x+C
$$
$$
\int \sec x\,\mathrm{d}x=\int \frac{\sec^2 x+\sec x\tan x}{\sec x+\tan x}\,\mathrm{d}x=\int \frac{d(\sec x+\tan x)}{\sec x+\tan x}=\ln\lvert \sec x+\tan x\rvert  +C
$$
$$
\int \csc x\,\mathrm{d}x=\int \frac{\csc^2x+\csc x\cot x}{\csc x+\cot x}\,\mathrm{d}x=\int \frac{-d(\csc x+\cot x)}{\csc x+\cot x}=-\ln\lvert \csc x+\cot x\rvert +C
$$
$$
\int \tan x\,\mathrm{d}x=\int \frac{\sin x}{\cos x}\,\mathrm{d}x=\int - \frac{\mathrm{d}\cos x}{\cos x}=-\ln \lvert \cos x\rvert +C
$$
$$
\int \cot x\,\mathrm{d}x=\int \frac{\cos x}{\sin x}\,\mathrm{d}x=\int \frac{\mathrm{d}\sin x}{\sin x}=\ln\lvert \sin x\rvert +C
$$
