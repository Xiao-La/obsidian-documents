$$
\begin{aligned}
P &= \boldsymbol{\tau}\cdot \mathbf{v}_{y}\ \\
&= \tau \frac{\partial y}{\partial t}\sin\theta \\
&\simeq - \tau \frac{\partial y}{\partial t} \frac{\partial y}{\partial x} \\
&=\tau\omega ky_{m}^2 \cos^2(kx-\omega t) \\

\end{aligned}
$$

$$
\begin{aligned}
\implies P_{\mathrm{avg}}&=\frac{1}{2}\tau\omega ky_{m}^2 \\
&=\frac{1}{2} \mu v^2 \omega ky_{m}^2 \\
&=\frac{1}{2} \mu v\omega^2 y_{m}^2
\end{aligned}
$$
